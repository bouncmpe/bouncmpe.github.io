{{/* Featured news carousel. Hextra ships no carousel component, so this is a small self-contained one: a scroll-snap track (which works on its own if scripting is unavailable) plus the arrows, dots and autoplay wired up by the inline script below. Slides are the news posts with `featured: true` in their front matter. */}} {{- $featured := where (where site.RegularPages.ByDate.Reverse "Section" "news") "Params.featured" true -}} {{- with $featured }} {{- $settings := site.Params.newsroom.image_settings.hero | default "1400x700 webp q90 Smart" -}}
{{- range $index, $page := . }} {{- $thumbnail := "" -}} {{- with $page.Params.thumbnail -}} {{- if reflect.IsMap . -}}{{- $thumbnail = .url -}}{{- else -}}{{- $thumbnail = . -}}{{- end -}} {{- end -}} {{- $src := "" -}} {{- if $thumbnail -}} {{- if hasPrefix $thumbnail "http" -}} {{- with try (resources.GetRemote $thumbnail) -}} {{- with .Value -}}{{- $src = (.Fill $settings).RelPermalink -}}{{- end -}} {{- end -}} {{- else -}} {{- with resources.Get $thumbnail -}}{{- $src = (.Fill $settings).RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- with $src }} {{- end }} {{ $page.Title }} {{- end }}
{{- range $index, $page := . }} {{- end }}
{{- end -}}
